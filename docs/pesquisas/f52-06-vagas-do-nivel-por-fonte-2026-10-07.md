## Consulta

Só leitura, por `psql` no contêiner do Postgres da stack de dev. As outras fontes não têm
tabela por fonte aqui; a tabela por tipo cobre todas.

```sql
-- Open, non-duplicate jobs in target areas at JUNIOR or MID.
select now() at time zone 'utc' as measured_at_utc;

select seniority, count(*)
from opportunities.opportunity o
where o.role_family in ('SOFTWARE_ENGINEERING', 'DATA')
  and o.lifecycle_status in ('DISCOVERED', 'ACTIVE')
  and o.duplicate_of is null
group by 1 order by 2 desc;

-- Per source type (a job with two sources counts in both).
select d.source_type,
       count(distinct o.id) filter (where o.seniority in ('JUNIOR', 'MID')) as jr_mid,
       count(distinct o.id) as target_area,
       count(distinct o.id) filter (where o.seniority in ('JUNIOR', 'MID') and o.work_mode = 'REMOTE') as jr_mid_remote,
       count(distinct d.id) filter (where d.enabled) as enabled_sources
from acquisition.source_definition d
left join opportunities.source_occurrence so on so.source_definition_id = d.id
left join opportunities.opportunity o on o.id = so.opportunity_id
  and o.role_family in ('SOFTWARE_ENGINEERING', 'DATA')
  and o.lifecycle_status in ('DISCOVERED', 'ACTIVE')
  and o.duplicate_of is null
group by 1 order by 2 desc;

-- inHire, per source.
select d.configuration->>'tenant_identifier' as tenant,
       count(distinct o.id) filter (where o.seniority in ('JUNIOR', 'MID')) as jr_mid,
       count(distinct o.id) as target_area,
       count(distinct so.opportunity_id) as all_jobs
from acquisition.source_definition d
left join opportunities.source_occurrence so on so.source_definition_id = d.id
left join opportunities.opportunity o on o.id = so.opportunity_id
  and o.role_family in ('SOFTWARE_ENGINEERING', 'DATA')
  and o.lifecycle_status in ('DISCOVERED', 'ACTIVE')
  and o.duplicate_of is null
where d.source_type = 'inhire'
group by 1 order by 2 desc, 3 desc, 1;
```
